rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function isBuyerOwner(buyerId) {
      let buyerDoc = get(/databases/$(database)/documents/buyers/$(buyerId));
      return isAuthenticated() && buyerDoc != null && (
        buyerDoc.data.get('auth_uid', '') == request.auth.uid || 
        buyerDoc.data.get('email', '').lower() == request.auth.token.get('email', '').lower() || 
        buyerDoc.data.get('google_email', '').lower() == request.auth.token.get('email', '').lower()
      );
    }

    function isSellerOwner(sellerId) {
      let sellerDoc = get(/databases/$(database)/documents/sellers/$(sellerId));
      return isAuthenticated() && sellerDoc != null && (
        sellerDoc.data.get('auth_uid', '') == request.auth.uid || 
        sellerDoc.data.get('email', '').lower() == request.auth.token.get('email', '').lower() || 
        sellerDoc.data.get('google_email', '').lower() == request.auth.token.get('email', '').lower()
      );
    }

    // Check if the user is the official admin
    function isAdmin() {
      return isAuthenticated() && (
        request.auth.token.admin == true ||
        request.auth.token.email == 'admin@dnd.com' ||
        request.auth.token.email == 'contact@dndpurchase.com'
      );
    }

    // Buyers Collection
    match /buyers/{userId} {
      allow create: if true; // Registration
      allow read: if true;
      allow update: if (isAuthenticated() && (resource.data.get('auth_uid', '') == request.auth.uid || resource.data.get('email', '').lower() == request.auth.token.get('email', '').lower() || resource.data.get('google_email', '').lower() == request.auth.token.get('email', '').lower())) && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['id', 'verified', 'created_at', 'auth_uid', 'email']) || isAdmin();
      allow delete: if isAdmin();
    }

    // Sellers Collection
    match /sellers/{userId} {
      allow create: if true; // Registration
      allow read: if true;
      // Allow updates to profile fields (including categories and product_manufacturers) 
      // but prevent modification of sensitive authentication bindings.
      allow update: if (isAuthenticated() && (resource.data.get('auth_uid', '') == request.auth.uid || resource.data.get('email', '').lower() == request.auth.token.get('email', '').lower() || resource.data.get('google_email', '').lower() == request.auth.token.get('email', '').lower())) && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['id', 'verified', 'created_at', 'auth_uid', 'email']) || isAdmin();
      allow delete: if isAdmin();
    }

    // Products & Product Options Collection
    // - Anyone can read (for the marketplace/catalog)
    // - Only Authenticated Admins (via Admin Portal) can modify
    match /products/{productId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /product_options/{optionId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Locations Collection
    // - Anyone can read (for checkout dropdowns)
    // - Only Admins can modify states and districts
    match /locations/{stateId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Settings Collection
    // - Anyone can read (for global app settings)
    // - Only Admins can modify
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Inquiries Collection
    match /inquiries/{inquiryId} {
      allow read: if isAuthenticated();
      allow create: if isBuyerOwner(request.resource.data.buyer_id);
      allow update, delete: if isBuyerOwner(resource.data.buyer_id) || isAdmin();
    }

    // Inquiry Items Collection
    // - Items attached to inquiries
    match /inquiry_items/{itemId} {
      allow read: if isAuthenticated();
      allow write: if isAuthenticated() || isAdmin();
    }

    // Offers Collection
    match /offers/{offerId} {
      allow read: if isAuthenticated();
      allow create: if isSellerOwner(request.resource.data.seller_id);
      allow update: if isSellerOwner(resource.data.seller_id) || isBuyerOwner(get(/databases/$(database)/documents/inquiries/$(resource.data.inquiry_id)).data.buyer_id) || isAdmin();
      allow delete: if isSellerOwner(resource.data.seller_id) || isAdmin();
    }

    // Soft Deleted Inquiries Collection
    match /soft_deleted_inquiries/{inquiryId} {
      allow read: if isAdmin() || isBuyerOwner(resource.data.buyer_id);
      allow create: if isBuyerOwner(request.resource.data.deleted_by_user_id) || isAdmin();
      allow update, delete: if isAdmin();
    }

    // Default fallback - deny all other access
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
